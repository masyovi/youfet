   439→   439→              const json = JSON.parse(xhr.responseText)
   440→   440→              resolve(json)
   441→   441→            } catch {
   442→   442→              reject(new Error(`Invalid response: ${xhr.responseText.substring(0, 200)}`))
   443→   443→            }
   444→   444→          } else {

   446→   446→              const json = JSON.parse(xhr.responseText)

   449→   449→              reject(new Error(`Upload failed (HTTP ${xhr.status})`))
   450→   450→            }
   451→   451→          }

   456→   456→        })
   457→   457→

   460→   460→        })

   462→   462→        xhr.open('POST', uploadServerUrl)

   466→   466→      console.log('[Upload] Result:', JSON.stringify(uploadResult))
   467→   467→
   468→   468→      if (uploadResult.status !== 200 || !uploadResult.files || uploadResult.files.length === 0) {
   469→   469→        throw new Error('Upload completed but no file code returned')
   470→   470→      }
   471→   471→
   472→   472→      const filecode = uploadResult.files[0].filecode
   473→   473→      const embedUrl = `https://luluvid.com/e/${filecode}`
   474→   474→      const thumbnailUrl = `https://img.lulucdn.com/${filecode}.jpg`

   476→   476→      setUploadResult({ filecode, embedUrl, thumbnailUrl })

   478→   478→    } catch (error) {

   480→   480→      toast.error(error instanceof Error ? error.message : 'Upload failed')
   481→   481→    } finally {
   482→   482→      setUploading(false)
   483→   483→      setUploadProgress(0)

   485→   485→  }
