   391→  const handleFileUpload = async () => {
   392→    if (!uploadFile) {
   393→      toast.error('Please select a file first')
   394→      return
   395→    }
   396→    setUploading(true)
   397→    setUploadProgress(0)
   398→
   399→    try {
   400→      console.log('[Upload] Starting:', uploadFile.name, (uploadFile.size / 1024 / 1024).toFixed(1), 'MB')
   401→
   402→      // Step 1: Get upload server URL from our API (small payload, no file)
   403→      const serverRes = await fetch('/api/lulustream/upload-server', {
   404→        method: 'POST',
   405→      })
   406→      const serverData = await serverRes.json()
   407→      if (!serverRes.ok || !serverData.success) {
   408→        throw new Error(serverData.error || 'Failed to get upload server')
   409→      }
   410→
   411→      const uploadServerUrl = serverData.uploadServer as string
   412→      const apiKey = serverData.apiKey as string
   413→      console.log('[Upload] Server:', uploadServerUrl)
   414→
   415→      // Step 2: Upload file DIRECTLY from browser to LuluStream (bypasses our server size limit)
   416→      const luluForm = new FormData()
   417→      luluForm.append('key', apiKey)
   418→      luluForm.append('file', uploadFile)
   419→      if (uploadTitle) luluForm.append('file_title', uploadTitle)
   420→      luluForm.append('file_public', '1')
   421→
   422→      // Use XMLHttpRequest for real upload progress tracking
   423→      const uploadResult = await new Promise<{
   424→        status: number
   425→        files: { filecode: string; filename: string; status: string }[]
   426→      }>((resolve, reject) => {
   427→        const xhr = new XMLHttpRequest()
   428→
   429→        xhr.upload.addEventListener('progress', (e) => {
   430→          if (e.lengthComputable) {
   431→            const pct = Math.round((e.loaded / e.total) * 100)
   432→            setUploadProgress(pct)
   433→          }
   434→        })
   435→
   436→        xhr.addEventListener('load', () => {
   437→          if (xhr.status >= 200 && xhr.status < 300) {
   438→            try {
   439→              const json = JSON.parse(xhr.responseText)
   440→              resolve(json)
   441→            } catch {
   442→              reject(new Error(`Invalid response: ${xhr.responseText.substring(0, 200)}`))
   443→            }
   444→          } else {
   445→            try {
   446→              const json = JSON.parse(xhr.responseText)
   447→              reject(new Error(json.msg || `Upload failed (HTTP ${xhr.status})`))
   448→            } catch {
   449→              reject(new Error(`Upload failed (HTTP ${xhr.status})`))
   450→            }
   451→          }
   452→        })
   453→
   454→        xhr.addEventListener('error', () => {
   455→          reject(new Error('Network error while uploading to LuluStream'))
   456→        })
   457→
   458→        xhr.addEventListener('abort', () => {
   459→          reject(new Error('Upload cancelled'))
   460→        })
   461→
   462→        xhr.open('POST', uploadServerUrl)
   463→        xhr.send(luluForm)
   464→      })
   465→
   466→      console.log('[Upload] Result:', JSON.stringify(uploadResult))
   467→
   468→      if (uploadResult.status !== 200 || !uploadResult.files || uploadResult.files.length === 0) {
   469→        throw new Error('Upload completed but no file code returned')
   470→      }
   471→
   472→      const filecode = uploadResult.files[0].filecode
   473→      const embedUrl = `https://luluvid.com/e/${filecode}`
   474→      const thumbnailUrl = `https://img.lulucdn.com/${filecode}.jpg`
   475→
   476→      setUploadResult({ filecode, embedUrl, thumbnailUrl })
   477→      toast.success('Video uploaded to LuluStream!')
   478→    } catch (error) {
   479→      console.error('[Upload] Error:', error)
   480→      toast.error(error instanceof Error ? error.message : 'Upload failed')
   481→    } finally {
   482→      setUploading(false)
   483→      setUploadProgress(0)
   484→    }
   485→  }
